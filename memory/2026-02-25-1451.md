### 方案 1：Playwright 脚本（推荐）

```javascript
// 导出每日数据
const { chromium } = require('playwright');

(async () => {
  const browser = await chromium.connect('http://127.0.0.1:18800');
  const page = browser.contexts()[0].pages()[0];
  
  // 登录
  await page.goto('https://your-system.com/login');
  await page.fill('#username', 'user');
  await page.fill('#password', 'pass');
  await page.click('#login');
  
  // 导航到导出页面
  await page.click('#data-menu');
  await page.click('#export');
  
  // 选择日期范围
  await page.fill('#start-date', '2026-02-01');
  await page.fill('#end-date', '2026-02-25');
  
  // 导出
  const [download] = await Promise.all([
    page.waitForEvent('download'),
    page.click('#export-btn')
  ]);
  
  await download.saveAs('/path/to/data.xlsx');
  await browser.close();
})();
```
